-- CREATE TABLES
CREATE TABLE Employee (
    emp_id  INT PRIMARY KEY,
    name    VARCHAR(50),
    salary  INT,
    dept    VARCHAR(50)
);

CREATE TABLE Audit_Log (
    log_id     INT PRIMARY KEY AUTO_INCREMENT,
    emp_id     INT,
    action     VARCHAR(50),
    old_salary INT,
    new_salary INT,
    log_time   DATETIME
);

-- INSERT
INSERT INTO Employee VALUES (1, 'Alice',   50000, 'HR');
INSERT INTO Employee VALUES (2, 'Bob',     70000, 'IT');
INSERT INTO Employee VALUES (3, 'Charlie', 60000, 'Finance');

-- 1. BEFORE INSERT
DELIMITER //
CREATE TRIGGER before_insert_emp
BEFORE INSERT ON Employee
FOR EACH ROW
BEGIN
    IF NEW.salary < 0 THEN
        SET NEW.salary = 0;
    END IF;
END //
DELIMITER ;
INSERT INTO Employee VALUES (4, 'Diana', -5000, 'IT');
SELECT * FROM Employee;

-- 2. AFTER INSERT
DELIMITER //
CREATE TRIGGER after_insert_emp
AFTER INSERT ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Audit_Log(emp_id, action, new_salary, log_time)
    VALUES (NEW.emp_id, 'INSERT', NEW.salary, NOW());
END //
DELIMITER ;
INSERT INTO Employee VALUES (5, 'Eve', 45000, 'HR');
SELECT * FROM Audit_Log;

-- 3. BEFORE UPDATE
DELIMITER //
CREATE TRIGGER before_update_emp
BEFORE UPDATE ON Employee
FOR EACH ROW
BEGIN
    IF NEW.salary < OLD.salary THEN
        SET NEW.salary = OLD.salary;
    END IF;
END //
DELIMITER ;
UPDATE Employee SET salary = 30000 WHERE emp_id = 1;
SELECT * FROM Employee;

-- 4. AFTER UPDATE
DELIMITER //
CREATE TRIGGER after_update_emp
AFTER UPDATE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Audit_Log(emp_id, action, old_salary, new_salary, log_time)
    VALUES (NEW.emp_id, 'UPDATE', OLD.salary, NEW.salary, NOW());
END //
DELIMITER ;
UPDATE Employee SET salary = 75000 WHERE emp_id = 2;
SELECT * FROM Audit_Log;

-- 5. BEFORE DELETE
DELIMITER //
CREATE TRIGGER before_delete_emp
BEFORE DELETE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Audit_Log(emp_id, action, old_salary, log_time)
    VALUES (OLD.emp_id, 'DELETE', OLD.salary, NOW());
END //
DELIMITER ;
DELETE FROM Employee WHERE emp_id = 3;
SELECT * FROM Audit_Log;

-- SHOW ALL TRIGGERS
SHOW TRIGGERS;

-- DROP TRIGGER
DROP TRIGGER before_insert_emp;
