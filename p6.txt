-- CREATE USERS
CREATE USER 'Alice'@'localhost' IDENTIFIED BY 'alice123';
CREATE USER 'Bob'@'localhost'   IDENTIFIED BY 'bob123';

-- GRANT
GRANT SELECT ON Employee TO 'Alice'@'localhost';
GRANT SELECT, INSERT ON Employee TO 'Bob'@'localhost';
GRANT ALL PRIVILEGES ON Employee TO 'Alice'@'localhost';
FLUSH PRIVILEGES;

-- REVOKE
REVOKE INSERT ON Employee FROM 'Bob'@'localhost';
REVOKE ALL PRIVILEGES ON Employee FROM 'Alice'@'localhost';
FLUSH PRIVILEGES;
