CREATE TABLE Employee (
    emp_id     INT PRIMARY KEY,
    name       VARCHAR(100) NOT NULL,
    department VARCHAR(50),
    salary     DECIMAL(10,2),
    age        INT
);

DELETE FROM Employee WHERE department = 'Finance';

SELECT name, salary, salary * 12 AS annual_salary
FROM Employee;

SELECT name, salary, salary - (salary * 0.10) AS salary_after_tax
FROM Employee;

SELECT name, salary, salary * 0.05 AS bonus
FROM Employee;

SELECT name, salary, salary / 12 AS monthly_salary
FROM Employee;

-- Modulus (%): Check even or odd emp_id
SELECT emp_id, name, emp_id % 2 AS is_odd
FROM Employee;


SELECT name,
       salary,
       salary * 0.05          AS bonus,
       salary * 0.10          AS tax,
       salary + (salary*0.05) - (salary*0.10) AS net_salary
FROM Employee;
