// =============================================
// SELECT DATABASE
// =============================================
use company_db

// INSERT DATA
db.Employee.insertMany([
    { emp_id: 1, name: "Alice",   salary: 50000, dept: "HR"      },
    { emp_id: 2, name: "Bob",     salary: 70000, dept: "IT"      },
    { emp_id: 3, name: "Charlie", salary: 60000, dept: "Finance"  },
    { emp_id: 4, name: "Diana",   salary: 80000, dept: "IT"      },
    { emp_id: 5, name: "Eve",     salary: 45000, dept: "HR"      }
]);

// =============================================
// 1. AGGREGATION
// =============================================

// $match - Filter documents (like WHERE)
db.Employee.aggregate([
    { $match: { dept: "IT" } }
]);

// $group - Group documents (like GROUP BY)
db.Employee.aggregate([
    { $group: {
        _id: "$dept",
        total_salary: { $sum: "$salary" },
        avg_salary:   { $avg: "$salary" },
        max_salary:   { $max: "$salary" },
        min_salary:   { $min: "$salary" },
        total_emp:    { $sum: 1 }
    }}
]);

// $sort - Sort documents (1=ASC, -1=DESC)
db.Employee.aggregate([
    { $sort: { salary: -1 } }
]);

// $limit - Limit number of documents
db.Employee.aggregate([
    { $limit: 3 }
]);

// $project - Show specific fields
db.Employee.aggregate([
    { $project: {
        name: 1,
        salary: 1,
        _id: 0
    }}
]);

// $count - Count documents
db.Employee.aggregate([
    { $count: "total_employees" }
]);

// =============================================
// 2. AGGREGATION PIPELINE
// (Multiple stages combined)
// =============================================

// Match + Group + Sort
db.Employee.aggregate([
    { $match:  { salary: { $gt: 50000 } } },
    { $group:  { _id: "$dept", avg_sal: { $avg: "$salary" } } },
    { $sort:   { avg_sal: -1 } }
]);

// =============================================
// 3. INDEXING
// =============================================

// Create single index
db.Employee.createIndex({ emp_id: 1 });

// Create index on name
db.Employee.createIndex({ name: 1 });

// Create compound index (multiple fields)
db.Employee.createIndex({ dept: 1, salary: -1 });

// Create unique index
db.Employee.createIndex({ emp_id: 1 }, { unique: true });

// Show all indexes
db.Employee.getIndexes();

// Drop specific index
db.Employee.dropIndex({ name: 1 });

// Drop all indexes
db.Employee.dropIndexes();

// Check query performance with index
db.Employee.find({ emp_id: 1 }).explain("executionStats");
