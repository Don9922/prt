-- CREATE TABLE
CREATE TABLE Account (
    acc_id  INT PRIMARY KEY,
    name    VARCHAR(50),
    balance INT
);

-- INSERT
INSERT INTO Account VALUES (1, 'Alice', 50000);
INSERT INTO Account VALUES (2, 'Bob',   30000);

-- 1. COMMIT
START TRANSACTION;
UPDATE Account SET balance = balance - 10000 WHERE acc_id = 1;
UPDATE Account SET balance = balance + 10000 WHERE acc_id = 2;
SELECT * FROM Account;
COMMIT;
SELECT * FROM Account;

-- 2. ROLLBACK
START TRANSACTION;
UPDATE Account SET balance = balance - 50000 WHERE acc_id = 1;
SELECT * FROM Account;
ROLLBACK;
SELECT * FROM Account;
