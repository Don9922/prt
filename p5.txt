
CREATE TABLE Employee (
    emp_id INT PRIMARY KEY,
    name   VARCHAR(50),
    email  VARCHAR(50),
    city   VARCHAR(50)
);

-- PATTERN MATCHING
SELECT * FROM Employee WHERE name LIKE 'A%';
SELECT * FROM Employee WHERE name LIKE '%n';
SELECT * FROM Employee WHERE name LIKE '%ar%';
SELECT * FROM Employee WHERE email LIKE '%@gmail.com';

-- STRING FUNCTIONS
SELECT name, UPPER(name)          AS upper_name   FROM Employee;
SELECT name, LOWER(name)          AS lower_name   FROM Employee;
SELECT name, LENGTH(name)         AS name_length  FROM Employee;
SELECT name, SUBSTRING(name,1,5)  AS short_name   FROM Employee;
SELECT CONCAT(name,' - ',city)    AS name_city    FROM Employee;
SELECT name, REPLACE(email,'@gmail.com','@company.com') AS new_email FROM Employee;
